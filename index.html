window.placeOrder = async function () { const user = auth.currentUser; if (!user) { alert("Pehle Login karo."); openLogin(); return; } if (!cartData.length) { alert("Cart empty hai."); return; } const customerName = prompt("Apna naam likho:"); if (!customerName) return; const phone = prompt("Mobile number likho:"); if (!phone) return; const address = prompt("Delivery address likho:"); if (!address) return; const total = cartData.reduce( (sum, food) => sum + Number(food.price), 0 ); try { await addDoc(collection(db, "orders"), { userUid: user.uid, customerEmail: user.email, customerName: customerName, phone: phone, address: address, items: cartData.map(food => ({ name: food.name, restaurant: food.restaurant, price: Number(food.price) })), total: total, paymentMethod: "COD", status: "Pending", createdAt: new Date().toISOString() }); alert( "✅ Order successfully placed!\n\n" + "Total: ₹" + total + "\nPayment: COD\nStatus: Pending" ); cartData = []; renderCart(); closeCart(); } catch (error) { alert( "❌ Order save nahi hua.\n\n" + error.message ); } };
